{#- Sidebar link renderer for the generated navigation tree (data/generated/navigation.json, built by scripts/dossier/build-navigation.mjs). One macro for all three depths so indentation, icon size and active state stay consistent as the tree grows: depth 0 = static top-level item, depth 1 = a dossier, depth 2 = one of that dossier's registries. `label` overrides the item's own label (used for a dossier's "Přehled dossieru" self-link inside its expanded subtree). -#} {#- `metrics` is data/generated/navigation-metrics.json (built by scripts/data/build-navigation-metrics.mjs). The macro takes the WHOLE map and resolves item.countMetric itself, so desktop and mobile cannot end up resolving the same item differently — there is one lookup, in one place. Pass metrics={} to render without badges. -#} {%- macro nav_link(item, depth, active_id, label="", metrics="") -%} {%- set is_active = item.id == active_id -%} {%- set pad = "px-4" -%} {%- set size = 20 -%} {%- if depth == 1 -%}{%- set pad = "pl-12 pr-4 text-[0.8rem]" -%}{%- set size = 16 -%}{%- endif -%} {%- if depth == 2 -%}{%- set pad = "pl-14 pr-4 text-[0.8rem]" -%}{%- set size = 16 -%}{%- endif -%} {#- `anchor` is for nodes that have no page of their own (a concept group is a section of /koncepty/, not a route) — the tree keeps them clickable. -#} {%- set anchor = item.anchor | default(value="") -%} {%- set href = get_url(path=item.path) -%} {%- if anchor -%}{%- set href = href ~ "#" ~ anchor -%}{%- endif -%} {%- set text = label | default(value="") -%} {%- if not text -%}{%- set text = item.label -%}{%- endif -%} {#- Resolve the named metric. A missing key renders no badge rather than a zero: absent and zero are different facts, and only the build-time validator is allowed to decide a declared metric is broken. -#} {%- set metric_id = item.countMetric | default(value="") -%} {%- set has_count = false -%} {%- set count = 0 -%} {%- if metric_id and metrics and metrics.metrics is defined -%} {%- if metrics.metrics[metric_id] is defined -%} {%- set_global has_count = true -%} {%- set_global count = metrics.metrics[metric_id].value -%} {%- endif -%} {%- endif -%} {%- set noun = item.countNoun | default(value="položek") -%} {#- The count also goes in `title`, which is what stays reachable when the sidebar is collapsed to icons and the badge itself is hidden. -#} {{ text }} {%- if has_count %} {#- aria-hidden: the number is already in the link's aria-label, so a screen reader must not read it twice. -#} {{ count }} {%- else %} {%- endif %} {%- endmacro nav_link -%}
