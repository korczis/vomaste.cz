{#- Sidebar link renderer for the generated navigation tree (data/generated/navigation.json, built by scripts/dossier/build-navigation.mjs). One macro for all three depths so indentation, icon size and active state stay consistent as the tree grows: depth 0 = static top-level item, depth 1 = a dossier, depth 2 = one of that dossier's registries. `label` overrides the item's own label (used for a dossier's "Přehled dossieru" self-link inside its expanded subtree). -#} {%- macro nav_link(item, depth, active_id, label="") -%} {%- set is_active = item.id == active_id -%} {%- set pad = "px-4" -%} {%- set size = 20 -%} {%- if depth == 1 -%}{%- set pad = "pl-12 pr-4 text-[0.8rem]" -%}{%- set size = 16 -%}{%- endif -%} {%- if depth == 2 -%}{%- set pad = "pl-14 pr-4 text-[0.8rem]" -%}{%- set size = 16 -%}{%- endif -%} {% if label %}{{ label }}{% else %}{{ item.label }}{% endif %} {%- endmacro nav_link -%}
